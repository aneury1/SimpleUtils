find_package(GTest CONFIG QUIET)
if(NOT GTest_FOUND)
    if(NOT UIUTILS_FETCH_GTEST)
        message(FATAL_ERROR "Install GoogleTest or enable UIUTILS_FETCH_GTEST")
    endif()
    include(FetchContent)
    # Immutable GoogleTest revision from its official CMake quickstart.
    FetchContent_Declare(googletest
        GIT_REPOSITORY https://github.com/google/googletest.git
        GIT_TAG 03597a01ee50ed33e9dfd640b249b4be3799d395)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
endif()
add_executable(uiutils_unit_tests style_tests.cpp restaurant_tests.cpp image_source_tests.cpp)
target_compile_features(uiutils_unit_tests PRIVATE cxx_std_17)
target_include_directories(uiutils_unit_tests PRIVATE ../include ../sample)
target_link_libraries(uiutils_unit_tests PRIVATE GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(uiutils_unit_tests)
if(UIUTILS_BUILD_GUI_TESTS)
    if(NOT UIUTILS_BUILD_LIBRARY)
        message(FATAL_ERROR "GUI tests require UIUTILS_BUILD_LIBRARY=ON")
    endif()
    add_executable(uiutils_gui_tests gui_tests.cpp)
    target_compile_features(uiutils_gui_tests PRIVATE cxx_std_17)
    target_compile_definitions(uiutils_gui_tests PRIVATE
        UIUTILS_TEST_PDF="${CMAKE_CURRENT_SOURCE_DIR}/../sample/assets/example.pdf")
    target_include_directories(uiutils_gui_tests PRIVATE ../src ${wxWidgets_INCLUDE_DIRS})
    target_compile_definitions(uiutils_gui_tests PRIVATE ${wxWidgets_DEFINITIONS})
    foreach(definition IN LISTS wxWidgets_DEFINITIONS_DEBUG)
        target_compile_definitions(uiutils_gui_tests PRIVATE $<$<CONFIG:Debug>:${definition}>)
    endforeach()
    if(wx_flags)
        target_compile_options(uiutils_gui_tests PRIVATE ${wx_flags})
    endif()
    target_link_libraries(uiutils_gui_tests PRIVATE uiutils::uiutils GTest::gtest)
    add_test(NAME uiutils_gui_tests COMMAND uiutils_gui_tests)
    set_tests_properties(uiutils_gui_tests PROPERTIES LABELS "gui")
endif()
