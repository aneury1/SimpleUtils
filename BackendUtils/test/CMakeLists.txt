set(SIMPLEUTILS_GTEST_SOURCE_DIR "" CACHE PATH "Local GoogleTest source tree (offline builds)")
if(SIMPLEUTILS_GTEST_SOURCE_DIR)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  add_subdirectory(${SIMPLEUTILS_GTEST_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/googletest EXCLUDE_FROM_ALL)
else()
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    if(NOT SIMPLEUTILS_FETCH_GTEST)
      message(FATAL_ERROR "GoogleTest not found. Install it, set SIMPLEUTILS_GTEST_SOURCE_DIR, or enable SIMPLEUTILS_FETCH_GTEST.")
    endif()
    include(FetchContent)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.tar.gz)
    FetchContent_MakeAvailable(googletest)
  endif()
endif()
add_executable(simpleutils_tests strings_test.cpp utilities_test.cpp encryption_test.cpp network_test.cpp misc_test.cpp odr_test.cpp)
if(TARGET GTest::gtest_main)
  target_link_libraries(simpleutils_tests PRIVATE simpleutils::simpleutils GTest::gtest_main)
else()
  target_link_libraries(simpleutils_tests PRIVATE simpleutils::simpleutils GTest::Main)
endif()
include(GoogleTest)
gtest_discover_tests(simpleutils_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 20)
